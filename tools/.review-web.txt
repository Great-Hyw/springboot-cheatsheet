
######## post-mapping @PostMapping [web/入门] org.springframework.web.bind.annotation.PostMapping
@Target METHOD
since=Spring 4.3 versionNote=-
tagline: POST 提交的快捷映射
keyPoints:
 - 等价于 @RequestMapping(method = RequestMethod.POST)，只匹配 POST。
 - 表单和 JSON 都能收，但接收方式不同：表单用对象参数或 @ModelAttribute，JSON 用 @RequestBody。
 - 创建资源想返回 201 + Location 时，用 ResponseEntity 或 @ResponseStatus(HttpStatus.CREATED)。
 - 同一个路径可以和 GET 方法共存，Spring 按 HTTP 方法区分。
notes:
 - POST 不幂等，客户端重试要靠幂等键或去重表，别指望框架兜底。
 - 浏览器地址栏发不了请求体，GET 请求里拿不到 @RequestBody。
 - 表单提交（application/x-www-form-urlencoded）用 @RequestBody 收不到，要用对象参数绑定。
attrs: name:String=""; value:String[]="{}"; path:String[]="{}"; params:String[]="{}"; headers:String[]="{}"; consumes:String[]="{}"; produces:String[]="{}"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/PostMapping.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-requestmapping.html
example:
import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class OrderController {

    // POST /orders，请求体是 JSON
    @PostMapping("/orders")
    @ResponseStatus(HttpStatus.CREATED)
    public Order create(@RequestBody CreateOrder cmd) {
        return new Order(1001L, cmd.productId(), cmd.qty());
    }

    public record CreateOrder(Long productId, int qty) {
    }

    public record Order(Long id, Long productId, int qty) {
    }
}


######## put-mapping @PutMapping [web/入门] org.springframework.web.bind.annotation.PutMapping
@Target METHOD
since=Spring 4.3 versionNote=-
tagline: 整体更新用 PUT 映射
keyPoints:
 - 等价于 @RequestMapping(method = RequestMethod.PUT)，只匹配 PUT。
 - PUT 按规范是幂等的：同一个请求重复发送，结果应保持一致。
 - 更新成功通常返回 200 或 204（无内容），不要默认依赖 302。
 - 路径里的 id 用 @PathVariable 取，请求体用 @RequestBody 接。
notes:
 - PUT 是整体替换：请求体没带的字段应当被清空，不想清空就改用 PATCH。
 - 路径 id 与请求体 id 可能不一致，以路径为准并显式校验。
 - 更新不存在的资源要返回 404，不要悄悄新建一条。
attrs: name:String=""; value:String[]="{}"; path:String[]="{}"; params:String[]="{}"; headers:String[]="{}"; consumes:String[]="{}"; produces:String[]="{}"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/PutMapping.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-requestmapping.html
example:
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class UserController {

    // PUT /users/7，整体替换
    @PutMapping("/users/{id}")
    public ResponseEntity<User> replace(@PathVariable Long id,
                                        @RequestBody User form) {
        return ResponseEntity.ok(new User(id, form.name(), form.email()));
    }

    public record User(Long id, String name, String email) {
    }
}


######## delete-mapping @DeleteMapping [web/入门] org.springframework.web.bind.annotation.DeleteMapping
@Target METHOD
since=Spring 4.3 versionNote=-
tagline: 删除资源用 DELETE
keyPoints:
 - 等价于 @RequestMapping(method = RequestMethod.DELETE)，只匹配 DELETE。
 - 成功且无响应体时返回 204 No Content，用 ResponseEntity.noContent() 最直接。
 - DELETE 也是幂等的：删不存在的资源返回 404 或 204 都行，但团队内要统一。
 - 浏览器表单只支持 GET/POST，前端删除要用 fetch 或 axios 发 DELETE。
notes:
 - 路径变量类型不匹配（比如 id 传字母）时，Spring 直接返回 400，不是 404。
 - 删除前要校验权限与数据归属，别仅凭 URL 里的 id 就删。
 - DELETE 带请求体不是普遍约定，部分代理会丢弃，尽量用路径或查询参数。
attrs: name:String=""; value:String[]="{}"; path:String[]="{}"; params:String[]="{}"; headers:String[]="{}"; consumes:String[]="{}"; produces:String[]="{}"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/DeleteMapping.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-requestmapping.html
example:
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;

@RestController
public class FileController {

    // DELETE /files/99
    @DeleteMapping("/files/{id}")
    public ResponseEntity<Void> remove(@PathVariable Long id) {
        return ResponseEntity.noContent().build();
    }

    // 批量删除：DELETE /files?ids=1,2,3
    @DeleteMapping("/files")
    public ResponseEntity<Void> removeBatch(@RequestParam List<Long> ids) {
        return ResponseEntity.noContent().build();
    }
}


######## request-param @RequestParam [web/入门] org.springframework.web.bind.annotation.RequestParam
@Target PARAMETER
since=Spring 2.5 versionNote=-
tagline: 绑定查询参数和表单字段
keyPoints:
 - name 和 value 互为别名；不写时按编译期参数名推断，需 -parameters 或显式指定。
 - required 默认 true，缺参数抛 MissingServletRequestParameterException 变成 400；设了 defaultValue 会自动变成 false。
 - defaultValue 是字符串，Spring 会按目标类型做类型转换。
 - 它绑定查询参数与表单数据，不是 JSON 请求体。
notes:
 - 参数名依赖编译期 -parameters，没有该选项时请显式写 name，否则可能取到 arg0。
 - defaultValue 写字符串也一样会做类型转换，比如 defaultValue = "1"。
 - 用 @RequestParam 收 JSON 一定拿不到值，那是 @RequestBody 的活。
attrs: name:String=""; value:String[]="{}"; defaultValue:String="ValueConstants.DEFAULT_NONE"; required:boolean="true"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/RequestParam.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-methods/requestparam.html
example:
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class SearchController {

    // GET /search?keyword=spring&page=1&size=20
    @GetMapping("/search")
    public String search(@RequestParam(name = "keyword") String keyword,
                         @RequestParam(defaultValue = "1") int page,
                         @RequestParam(defaultValue = "20") int size) {
        return "keyword=" + keyword + ", page=" + page + ", size=" + size;
    }

    // 可选参数：不传也能进方法
    @GetMapping("/filter")
    public String filter(@RequestParam(required = false) String category) {
        return "category=" + category;
    }
}


######## response-body @ResponseBody [web/入门] org.springframework.web.bind.annotation.ResponseBody
@Target TYPE, METHOD
since=Spring 3.0 versionNote=-
tagline: 直接把返回值写进响应体
keyPoints:
 - 方法级优先，类级对该类所有方法生效。
 - 序列化格式由 produces 和消息转换器决定，默认 JSON（有 Jackson 时）。
 - 返回 String 时默认 text/plain，不是 HTML；要返回 HTML 得显式配 produces。
 - 与 @ResponseStatus、ResponseEntity 搭配可以控制状态码和响应头。
notes:
 - 在 @RestController 类里再写 @ResponseBody 是多余的。
 - String 返回值容易被误认为页面名：想返回 HTML 要设置 produces 为 text/html。
 - 返回值是 null 时默认 200 且响应体为空，不是 404。
attrs: 
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/ResponseBody.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-methods/responsebody.html
example:
import org.springframework.stereotype.Controller;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.ResponseBody;

import java.util.Map;

@Controller
public class LegacyController {

    // 方法级 @ResponseBody：返回 Map 序列化成 JSON
    @GetMapping("/legacy/ping")
    @ResponseBody
    public Map<String, Object> ping() {
        return Map.of("ok", true, "msg", "pong");
    }

    // 同一个类里仍可返回视图名，走模板解析
    @GetMapping("/legacy/page")
    public String page() {
        return "legacy/page";
    }
}


######## request-header @RequestHeader [web/入门] org.springframework.web.bind.annotation.RequestHeader
@Target PARAMETER
since=Spring 3.0 versionNote=-
tagline: 读取请求头里的值
keyPoints:
 - name/value 指定请求头名；required 默认 true；设了 defaultValue 会自动变成 false。
 - 请求头名大小写不敏感（HTTP 规范），代码里建议统一写法。
 - 参数类型也可以是 Map<String, String> 或 HttpHeaders，一次拿到全部请求头。
 - 多值头请用 List 或 HttpHeaders 接收，别硬塞进单个 String。
notes:
 - 请求头名写错不会报找不到方法，而是 400 参数缺失，排查时容易懵。
 - 自定义头要进网关和 CORS 白名单，否则浏览器端可能带不过来。
 - Authorization 头别直接打进日志，注意脱敏。
attrs: name:String=""; value:String[]="{}"; defaultValue:String="ValueConstants.DEFAULT_NONE"; required:boolean="true"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/RequestHeader.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-methods/requestheader.html
example:
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RestController;

import java.util.Map;

@RestController
public class HeaderController {

    // 必填头缺失 → 400
    @GetMapping("/tenant")
    public String tenant(@RequestHeader("X-Tenant-Id") String tenantId) {
        return "tenant=" + tenantId;
    }

    // 可选头，带默认值
    @GetMapping("/lang")
    public String lang(@RequestHeader(name = "Accept-Language", defaultValue = "zh-CN") String lang) {
        return "lang=" + lang;
    }

    // 一次拿全部请求头
    @GetMapping("/headers")
    public Map<String, String> all(@RequestHeader Map<String, String> headers) {
        return headers;
    }
}


######## cookie-value @CookieValue [web/入门] org.springframework.web.bind.annotation.CookieValue
@Target PARAMETER

import org.springframework.web.bind.annotation.CookieValue;

import org.springframework.web.bind.annotation.RestController;

@RestController

    // 浏览器带 Cookie: SESSION=abc123
    @GetMapping("/profile")

        return "theme=" + theme;
    }
}


######## controller-advice @ControllerAdvice [web/进阶] org.springframework.web.bind.annotation.ControllerAdvice
@Target TYPE
since=Spring 3.2 versionNote=-
tagline: 全局增强所有控制器
keyPoints:
 - 默认对所有控制器生效；用 basePackages、basePackageClasses、assignableTypes、annotations 精确限定范围。
 - name 是 Bean 名称别名（Spring 6.1+），value 是 basePackages 的别名，一般不用写。
 - 里面的 @ExceptionHandler 优先级低于控制器本地的异常处理方法。
 - 返回视图的全局处理用它；返回 JSON 用 @RestControllerAdvice 更省事。
notes:
 - 它不会自动处理异常，必须配合 @ExceptionHandler 才有行为。
 - 本地控制器已有同类型 @ExceptionHandler 时，advice 里的不会生效。
 - 范围写错（比如 basePackages 拼错）会导致完全不生效，启动日志里也不会报错。
attrs: annotations:Class<? extends Annotation>[]="{}"; name:String=""; value:String[]="{}"; basePackages:String[]="{}"; assignableTypes:Class<?>[]="{}"; basePackageClasses:Class<?>[]="{}"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/ControllerAdvice.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-advice.html
example:
import org.springframework.stereotype.Controller;
import org.springframework.web.bind.annotation.ControllerAdvice;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.ModelAttribute;
import org.springframework.web.servlet.ModelAndView;

// 对所有 @Controller 生效
@ControllerAdvice
public class GlobalControllerAdvice {

    // 每个控制器方法执行前都往模型里放站点信息
    @ModelAttribute("siteName")
    public String siteName() {
        return "小明的技术站";
    }

    // 未捕获的 IllegalStateException 统一渲染错误页
    @ExceptionHandler(IllegalStateException.class)
    public ModelAndView handleState(IllegalStateException ex) {
        ModelAndView mav = new ModelAndView("error/500");
        mav.addObject("message", ex.getMessage());
        return mav;
    }
}


######## exception-handler @ExceptionHandler [web/进阶] org.springframework.web.bind.annotation.ExceptionHandler
@Target METHOD
since=Spring 3.0 versionNote=-
tagline: 把异常转成友好响应
keyPoints:
 - value 与 exception 互为别名（exception 为 Spring 6.2 新增），列出要处理的异常类型。
 - 方法参数可注入异常对象、WebRequest、HttpServletRequest 等，返回值走正常的消息转换。
 - produces 可限制响应的 Content-Type（Spring 6.2 新增）。
 - 按异常类型的精确程度匹配：先找最具体的类型，父类处理方法兜底。
notes:
 - 处理方法的返回类型若和 @ResponseStatus 冲突，状态码可能不符合预期，要测。
 - 同一异常类型在本地和全局都有处理时，本地优先。
 - 别捕获 Throwable 或 Exception 后返回 200，要给出对应状态码。
attrs: value:Class<? extends Throwable>[]="{}"; exception:Class<? extends Throwable>[]="{}"; produces:String[]="{}"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/ExceptionHandler.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-exceptionhandler.html
example:
import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class ItemController {

    @GetMapping("/items/{id}")
    public String item(@PathVariable Long id) {
        if (id > 100) {
            throw new ItemNotFoundException(id);
        }
        return "item-" + id;
    }

    // 只处理本控制器的 ItemNotFoundException
    @ExceptionHandler(ItemNotFoundException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public String notFound(ItemNotFoundException ex) {
        return "找不到商品：" + ex.getId();
    }

    public static class ItemNotFoundException extends RuntimeException {
        private final Long id;
        public ItemNotFoundException(Long id) {
            super("item " + id);
            this.id = id;
        }
        public Long getId() { return id; }
    }
}


######## response-status @ResponseStatus [web/入门] org.springframework.web.bind.annotation.ResponseStatus
@Target TYPE, METHOD
since=Spring 3.0 versionNote=-
tagline: 声明响应状态码
keyPoints:
 - 标在方法上：成功时用该状态码；标在异常类上：该异常抛出时用该状态码。
 - value 与 code 互为别名，默认 INTERNAL_SERVER_ERROR。
 - reason 非空时容器直接渲染错误页，方法返回值被忽略，前后端分离项目通常不用它。

attrs: value:HttpStatus="500 INTERNAL_SERVER_ERROR"; code:HttpStatus="500 INTERNAL_SERVER_ERROR"; reason:String=""
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/web/bind/annotation/ResponseStatus.html | https://docs.spring.io/spring-framework/reference/web/webmvc/mvc-controller/ann-exceptionhandler.html

import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class ArticleController {

    // 创建成功返回 201 Created
    @PostMapping("/articles")
    @ResponseStatus(HttpStatus.CREATED)
    public String create() {
        return "created";
    }

    // GET /articles/999 抛异常后返回 404
    @GetMapping("/articles/{id}")
    public String detail(@PathVariable Long id) {
        if (id > 500) {
            throw new ArticleNotFoundException();
        }
        return "article-" + id;
    }

    // 异常类上声明状态码，抛出即生效
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public static class ArticleNotFoundException extends RuntimeException {
    }
}


######## valid @Valid [validation/入门] jakarta.validation.Valid
@Target METHOD, FIELD, CONSTRUCTOR, PARAMETER, TYPE_USE
since=Jakarta Bean Validation 3.0 versionNote=-
tagline: 触发级联校验的标记
keyPoints:
 - 它是级联标记，不是约束；单独写它不会校验任何东西。
 - 可标在参数、字段、方法、构造器和类型使用处；集合元素上写 @Valid 可逐个校验。

import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

public class UserController {

    }

    public record CreateUser(

            @Min(value = 18, message = "年龄至少 18") int age) {
    }
}


######## validated @Validated [validation/进阶] org.springframework.validation.annotation.Validated
@Target TYPE, METHOD, PARAMETER
since=Spring 3.1 versionNote=-
tagline: Spring 的分组校验注解
keyPoints:
 - value 声明校验分组，如 @Validated(CreateGroup.class)；不写就是默认组。
 - 类级 @Validated 会启用方法校验（基于 AOP 代理），方法参数上加约束注解才会被检查。
 - 控制器方法参数上用它时，MVC 会按指定分组做数据绑定校验。
 - 它是 Spring 专有注解，脱离 Spring 环境不生效；纯 Bean Validation 用 @Valid。
notes:
 - @Validated 不写分组时等价于默认组，和 @Valid 效果接近，但包名完全不同。
 - 类级 @Validated 依赖 AOP 代理，同类内部自调用不会触发校验。
 - 方法校验失败抛 ConstraintViolationException，不是 MethodArgumentNotValidException，异常处理要分开写。
attrs: value:Class<?>[]="{}"
docs: https://docs.spring.io/spring-framework/docs/6.2.x/javadoc-api/org/springframework/validation/annotation/Validated.html | https://docs.spring.io/spring-framework/reference/core/validation/beanvalidation.html
example:
import jakarta.validation.constraints.NotBlank;

import org.springframework.stereotype.Service;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class SignupController {

    // 只校验 CreateGroup 分组里的规则
    @PostMapping("/signup")
    public String signup(@Validated(CreateGroup.class) @RequestBody SignupForm form) {
        return "ok " + form.username();
    }

    public interface CreateGroup {
    }

    public record SignupForm(
            @NotBlank(groups = CreateGroup.class) String username,
            String inviteCode) {
    }

    // 类级 @Validated：Spring 对方法参数做校验
    @Service
    @Validated
    public static class NameService {
        public String hello(@NotBlank String name) {
            return "hello " + name;
        }
    }
}


######## not-empty @NotEmpty [validation/入门] jakarta.validation.constraints.NotEmpty
@Target METHOD, FIELD, ANNOTATION_TYPE, CONSTRUCTOR, PARAMETER, TYPE_USE
since=Jakarta Bean Validation 3.0 versionNote=-
tagline: 集合/字符串不能为空
keyPoints:
 - 适用于 CharSequence、Collection、Map 和数组，其他类型会抛 UnexpectedTypeException。
 - 空集合不通过，但只含空白字符的字符串能通过（长度不为零）。

 - 给 String 用它挡不住全空格，文本场景应改用 @NotBlank。

######## size @Size [validation/入门] jakarta.validation.constraints.Size
@Target METHOD, FIELD, ANNOTATION_TYPE, CONSTRUCTOR, PARAMETER, TYPE_USE
since=Jakarta Bean Validation 3.0 versionNote=-
tagline: 限制长度或数量范围
keyPoints:
 - min 默认 0，max 默认 Integer.MAX_VALUE，可以只限制一端。
 - 对 String 数的是字符数（按 String.length()），对集合数的是元素个数。
 - 不支持基本类型和任意 POJO，用错类型会抛校验异常。
notes:
 - @Size 在 String 上是 null 也算通过，必填要另加 @NotBlank。
 - emoji 等补充字符按 UTF-16 会计两个长度，需要精确按码点要自己实现。
 - 只写 max 不写 min 时，空串也能通过，别误以为等于必填。
attrs: min:int="0"; max:int="2147483647"; message:String="{jakarta.validation.constraints.Size.message}"; groups:Class<?>[]="{}"; payload:Class<? extends Payload>[]="{}"
docs: https://jakarta.ee/specifications/bean-validation/3.0/apidocs/jakarta/validation/constraints/Size.html | https://jakarta.ee/specifications/bean-validation/3.0/jakarta-bean-validation-spec-3.0.html
example:
import jakarta.validation.Valid;
import jakarta.validation.constraints.Size;

import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;

@RestController
public class PostController {

    @PostMapping("/posts")
    public String create(@Valid @RequestBody PostForm form) {
        return "title len " + form.title().length();
    }

    public record PostForm(
            @Size(min = 1, max = 100, message = "标题 1 到 100 个字") String title,
            @Size(max = 500, message = "一次最多 500 条") List<Long> tagIds) {
    }
}


######## past @Past [validation/入门] jakarta.validation.constraints.Past
@Target METHOD, FIELD, ANNOTATION_TYPE, CONSTRUCTOR, PARAMETER, TYPE_USE
since=Jakarta Bean Validation 3.0 versionNote=-
tagline: 时间必须在过去
keyPoints:
 - 支持 Date、Calendar、LocalDate、LocalDateTime、Instant 等时间类型。
 - 较新版本支持以 ClockProvider 注入的时钟为基准判断，便于测试。
 - null 不校验，必填请叠加 @NotNull。

 - LocalDate 的“今天”与服务器时区有关，跨时区业务要统一时区。

import jakarta.validation.Valid;
import jakarta.validation.constraints.Past;

import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

import java.time.LocalDate;

@RestController
public class BirthController {

    @PostMapping("/birthdays")
    public String save(@Valid @RequestBody BirthForm form) {

    }

######## future @Future [validation/入门] jakarta.validation.constraints.Future
@Target METHOD, FIELD, ANNOTATION_TYPE, CONSTRUCTOR, PARAMETER, TYPE_USE
since=Jakarta Bean Validation 3.0 versionNote=-
tagline: 时间必须在未来
keyPoints:
 - 支持 Date、Calendar、LocalDate、LocalDateTime、Instant 等时间类型。
 - 判断基准来自 ClockProvider，可注入固定时钟做单元测试。
 - null 不校验，必填要配 @NotNull。

 - @Future 不含当前时刻，要求“现在起”需要用 @FutureOrPresent。
 - 客户端与服务器时区不同时，用户以为的“明天”可能已经是过去。
 - 定时任务补偿时，历史数据再走同一个接口会被 @Future 拦下。
attrs: message:String="{jakarta.validation.constraints.Future.message}"; groups:Class<?>[]="{}"; payload:Class<? extends Payload>[]="{}"

example:
import jakarta.validation.Valid;
import jakarta.validation.constraints.Future;
import jakarta.validation.constraints.NotNull;

import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

import java.time.LocalDateTime;

@RestController
public class BookingController {

    @PostMapping("/bookings")
    public String book(@Valid @RequestBody BookingForm form) {
        return "booked at " + form.startAt();
    }

    public record BookingForm(

            @Future(message = "预约时间必须在未来") LocalDateTime startAt) {
    }
}
